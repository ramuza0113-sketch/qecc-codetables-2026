\begin{abstract}
Using a Meet-in-the-Middle norm-matching construction in the group algebra $\mathbb{F}_{q^2}[x]/(x^m - 1)$ combined with GPU-accelerated minimum-weight screening, we construct ten explicit Hermitian self-orthogonal $[n, m]_{q^2}$ linear codes whose Hermitian duals achieve minimum Hamming weight $d(C^{\perp_H}) \ge d_{\mathrm{new}}$. By the Hermitian stabilizer construction of Ashikhmin--Knill and Ketkar \emph{et al.}, these yield ten pure $q$-ary quantum error-correcting codes:
$[[36, 26, 4]]_2$, $[[49, 33, 5]]_2$, $[[20, 10, 5]]_3$, $[[18, 6, 6]]_4$, $[[19, 7, 6]]_4$, $[[20, 10, 5]]_4$, $[[25, 15, 5]]_4$, $[[26, 16, 5]]_4$, $[[19, 7, 6]]_5$, and $[[21, 7, 7]]_7$.
Together with trivial stabilizer extension and pure subcode propagation, these constructions strictly improve \textbf{sixteen} minimum-distance lower bounds in Grassl's online tables (\texttt{CodeTables.de}) across local dimensions $q \in \{2, 3, 4, 5, 7\}$, closing \textbf{five} previously open entries ($[[36, 26, 4]]_2$, $[[37, 26, 4]]_2$, $[[20, 10, 5]]_3$, $[[25, 15, 5]]_4$, and $[[26, 16, 5]]_4$) to exact optimality. Every code is specified completely in closed polynomial form and verified from first principles by standalone zero-symmetry verifiers in Rust, Python, and Magma.
\end{abstract}

\section{Introduction}

Let $\mathbb{F}_q$ be the finite field with $q$ elements ($q$ a prime power). An $[[n, k, d]]_q$ quantum stabilizer code encodes $k$ logical $q$-ary qudits into $n$ physical qudits and corrects any error affecting at most $\lfloor (d-1)/2 \rfloor$ qudits (or detects up to $d-1$ errors)~\cite{Calderbank1998,Ashikhmin2001,Ketkar2006}. Determining the largest achievable minimum distance $d$ for given $(q, n, k)$ is a central problem in quantum coding theory; the current state of knowledge is maintained by Grassl in the reference database \texttt{CodeTables.de}~\cite{Grassl2007}.

A foundational algebraic route to stabilizer codes is the \emph{Hermitian construction}~\cite{Ashikhmin2001,Ketkar2006}: let $\mathbb{F}_{q^2}$ carry the Frobenius involution $\overline{u} = u^q$, and equip $\mathbb{F}_{q^2}^n$ with the Hermitian inner product
\[
  \langle \mathbf{u}, \mathbf{v} \rangle_H = \sum_{j=0}^{n-1} u_j v_j^q \in \mathbb{F}_{q^2}.
\]
If $C \subseteq \mathbb{F}_{q^2}^n$ is an $\mathbb{F}_{q^2}$-linear $[n, m]_{q^2}$ code that is \emph{Hermitian self-orthogonal} ($C \subseteq C^{\perp_H}$), then there exists an $[[n, n - 2m, d]]_q$ stabilizer code with minimum distance $d = \min\{\mathrm{wt}(\mathbf{v}) : \mathbf{v} \in C^{\perp_H} \setminus C\}$; when $d(C^{\perp_H}) \ge d_{\mathrm{new}}$, the resulting quantum code is \emph{pure} of minimum distance at least $d_{\mathrm{new}}$~\cite[Corollary~19]{Ketkar2006}.

In this note we combine systematic multi-circulant and bordered generator matrices over $\mathbb{F}_{q^2}$ with $O(1)$ hash-table norm matching in $\mathbb{F}_{q^2}[x]/(x^m - 1)$ and parallel forced-pivot distance screening on a consumer GPU (NVIDIA GeForce RTX 3090). This yields ten explicit pure quantum codes that improve sixteen entries in \texttt{CodeTables.de} across $q \in \{2, 3, 4, 5, 7\}$, establishing exact optimal minimum distances in five cases.

\section{Algebraic Construction and Meet-in-the-Middle Search}

Fix a block size $m \ge 3$ and let $R_m = \mathbb{F}_{q^2}[x]/(x^m - 1)$. With each polynomial $a(x) = \sum_{t=0}^{m-1} a_t x^t \in R_m$ we associate the $m \times m$ right-circulant matrix $M(a)$ whose $(r, c)$ entry ($0 \le r, c < m$) is $a_{(r - c) \bmod m}$ (column $0$ equals $(a_0, a_1, \dots, a_{m-1})^\top$). Define the conjugate-transpose involution on $R_m$ by
\[
  a^\dagger(x) = a_0^q + \sum_{t=1}^{m-1} a_{m-t}^q x^t.
\]
Then $M(a)^\dagger = \overline{M(a)}^\top = M(a^\dagger)$ and $M(a) M(b) = M(a b \bmod (x^m - 1))$.

\begin{proposition}[Multi-Circulant and Bordered Hermitian Self-Orthogonality]\label{prop:hermitian}
Let $\ell \ge 3$, let $a_1(x), \dots, a_{\ell-1}(x) \in R_m$, and let $\mathbf{1}_m = (1, \dots, 1)^\top \in \mathbb{F}_{q^2}^m$.
\begin{enumerate}
  \item \emph{(Unbordered, $n = \ell m$):} The $m \times \ell m$ matrix
  \[
    G = \bigl[\, I_m \;\big|\; M(a_1) \;\big|\; \cdots \;\big|\; M(a_{\ell-1}) \,\bigr]
  \]
  has rank $m$ over $\mathbb{F}_{q^2}$ and satisfies $G G^\dagger = \mathbf{0}$ if and only if
  \begin{equation}\label{eq:unbordered}
    \sum_{i=1}^{\ell-1} a_i(x)\, a_i^\dagger(x) \equiv -1 \pmod{x^m - 1}.
  \end{equation}
  \item \emph{(Bordered, $n = \ell m + 1$):} The $m \times (\ell m + 1)$ matrix
  \[
    G_{\mathrm{bord}} = \bigl[\, I_m \;\big|\; M(a_1) \;\big|\; \cdots \;\big|\; M(a_{\ell-1}) \;\big|\; \mathbf{1}_m \,\bigr]
  \]
  has rank $m$ over $\mathbb{F}_{q^2}$ and satisfies $G_{\mathrm{bord}} G_{\mathrm{bord}}^\dagger = \mathbf{0}$ if and only if
  \begin{equation}\label{eq:bordered}
    \sum_{i=1}^{\ell-1} a_i(x)\, a_i^\dagger(x) \equiv -1 - \sum_{t=0}^{m-1} x^t \pmod{x^m - 1}.
  \end{equation}
\end{enumerate}
\end{proposition}

\begin{proof}
Since $G$ begins with $I_m$, $\mathrm{rank}_{\mathbb{F}_{q^2}}(G) = m$ holds unconditionally. For~(1), $G G^\dagger = I_m + \sum_{i=1}^{\ell-1} M(a_i) M(a_i)^\dagger = M\bigl(1 + \sum_{i=1}^{\ell-1} a_i a_i^\dagger\bigr)$, which vanishes if and only if~\eqref{eq:unbordered} holds. For~(2), the border column contributes $\mathbf{1}_m \mathbf{1}_m^\dagger = J_m = M\bigl(\sum_{t=0}^{m-1} x^t\bigr)$ to $G_{\mathrm{bord}} G_{\mathrm{bord}}^\dagger$, yielding~\eqref{eq:bordered}.
\end{proof}

\paragraph{Subspace Collapse and $O(1)$ Meet-in-the-Middle Generation.}
For every $a \in R_m$, the Hermitian norm $N(a) = a(x) a^\dagger(x) \bmod (x^m - 1)$ satisfies $N(a)^\dagger = N(a)$: its constant term lies in $\mathbb{F}_q$ and $N(a)_{m-t} = N(a)_t^q$ for $1 \le t < m$. Consequently, $N(a)$ takes values in an $\mathbb{F}_q$-subspace of cardinality only $q^m$ (rather than $q^{2m}$). Indexing a pool $\mathcal{P} \subseteq R_m$ of polynomials of weight at least $d_{\mathrm{new}} - 1$ by $N(a)$ in a hash table allows $(\ell - 1)$-tuples solving~\eqref{eq:unbordered} or~\eqref{eq:bordered} to be sampled in $O(1)$ expected operations per exact Hermitian self-orthogonal code.

\paragraph{Forced-Pivot Minimum-Distance Screening.}
Write $G = [I_m \mid B]$ with $B \in \mathbb{F}_{q^2}^{m \times (n - m)}$. Because Frobenius conjugation preserves Hamming weight, $d(C^{\perp_H}) \ge d_{\mathrm{new}}$ holds if and only if every nonzero vector $\mathbf{v} = (\mathbf{v}_0 \mid \mathbf{v}_B) \in \ker(G)$ satisfies $\mathrm{wt}(\mathbf{v}_B) + \mathrm{wt}(B \mathbf{v}_B) \ge d_{\mathrm{new}}$. Cyclic shift invariance across the $\ell - 1$ blocks (which fixes $\mathbf{1}_m$ when bordered) restricts the first active coordinate of $\mathbf{v}_B$ to at most $\ell$ anchor columns with projective coefficient $1$. Moreover, when testing whether a partial sum $\mathbf{u} \ne \mathbf{0}$ of $d_{\mathrm{new}} - 2$ columns of $B$ can be cancelled by one final column $\alpha \mathbf{b}_j$ ($\mathbf{u} + \alpha \mathbf{b}_j = \mathbf{0}$), any nonzero coordinate $u_{r_0} \ne 0$ uniquely forces $\alpha = -u_{r_0} b_{j, r_0}^{-1}$, eliminating inner scalar loops on GPU threads.

\section{Summary of Improved Bounds and Explicit Defining Vectors}

\begin{itemize}
  \item \textbf{Extension ($\mathrm{Extend}$):} Any $[[n, k, d]]_q$ stabilizer code extends trivially to $[[n + s, k, \ge d]]_q$ ($s \ge 1$) by adjoining $s$ unentangled ancilla qudits stabilized by $Z_{n+1}, \dots, Z_{n+s}$.
  \item \textbf{Pure Subcode ($\mathrm{PureSubcode}$):} Any \emph{pure} $[[n, k, d]]_q$ stabilizer code with $k \ge 1$ yields an $[[n, k - 1, \ge d]]_q$ stabilizer code by adjoining any non-stabilizer centralizer generator in $S^{\perp_s} \setminus S$ (whose centralizer shrinks, preserving minimum weight $\ge d$).
\end{itemize}

\subsection{Field Encodings and Complete Defining Vectors}

To make every construction self-contained on paper, we fix the integer encoding $u \in \{0, \dots, q^2 - 1\}$ of $\mathbb{F}_{q^2}$ for each $q \in \{2, 3, 4, 5, 7\}$:
\begin{itemize}
  \item $q = 2$: $\mathbb{F}_4 = \mathbb{F}_2[\omega]/(\omega^2 + \omega + 1)$, encoded as $u = c_0 + 2 c_1$ ($c_0, c_1 \in \{0, 1\}$).
  \item $q \in \{3, 5\}$: $\mathbb{F}_{q^2} = \mathbb{F}_q[\theta]/(\theta^2 - 2)$, encoded as $u = x + q y$ ($x, y \in \{0, \dots, q-1\}$), so $\theta^q = -\theta$.
  \item $q = 4$: $\mathbb{F}_{16} = \mathbb{F}_2[z]/(z^4 + z + 1)$, encoded in binary as $u = \sum_{j=0}^{3} c_j 2^j$ ($c_j \in \{0, 1\}$), with $\overline{u} = u^4$.
  \item $q = 7$: $\mathbb{F}_{49} = \mathbb{F}_7[\theta]/(\theta^2 - 3)$, encoded as $u = x + 7 y$ ($x, y \in \{0, \dots, 6\}$), so $\theta^7 = -\theta$.
\end{itemize}

\section{Independent Clean-Room Verification and Data Availability}

Every code was verified by two completely independent zero-shared-code programs without assuming cyclic or quasi-twisted symmetry:
\begin{enumerate}
  \item \textbf{Multi-threaded Rust Exhaustive Syndrome Verifier (\texttt{verify\_cleanroom.rs}):} Reads only the raw stabilizer / Hermitian matrix, verifies $\mathrm{rank}_{\mathbb{F}_q}([X \mid Z]) = n - k$ and $X Z^\top - Z X^\top \equiv \mathbf{0} \pmod q$ (or $\mathrm{rank}_{\mathbb{F}_{16}}(G) = m$ and $G G^\dagger = \mathbf{0}$), and exhaustively enumerates all $\sum_{w=1}^{d_{\mathrm{new}}-1} \binom{n}{w} (q+1)(q^2-1)^{w-1}$ projective operators across 20 CPU threads ($24.66$ billion operators verified for the first nine codes with zero degenerate or logical operators below weight $d_{\mathrm{new}}$).
  \item \textbf{Projective Meet-in-the-Middle Verifier (\texttt{verify\_all\_qecc.py}) \& Magma Script (\texttt{magma/verify\_all\_10\_codes.m}):} Verifies SHA-256 digests, rank, Hermitian and symplectic self-orthogonality, and $d(C^{\perp_H}) \ge d_{\mathrm{new}}$ across all ten codes (including $[[21, 7, 7]]_7$) in $1.9$~seconds in Python, alongside a native Magma verification script calling \texttt{HermitianDual} and \texttt{QuantumCode}.
\end{enumerate}

All JSON certificates, SHA-256 checksums, Magma scripts, search solvers, and clean-room verifiers are publicly archived at \url{https://github.com/ramuza0113-sketch/qecc-codetables-2026}.
