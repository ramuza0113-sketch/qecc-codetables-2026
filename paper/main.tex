\begin{abstract}
Using a Meet-in-the-Middle norm-matching construction in the $\lambda$-constacyclic algebra $\mathbb{F}_{q^2}[x]/(x^m - \lambda)$ ($\lambda^{q+1} = 1$) combined with GPU-accelerated forced-pivot minimum-weight screening, we construct eighteen explicit Hermitian self-orthogonal $[n, m]_{q^2}$ linear codes whose Hermitian duals achieve minimum Hamming weight $d(C^{\perp_H}) \ge d_{\mathrm{new}}$. By the Hermitian stabilizer construction of Ashikhmin--Knill and Ketkar \emph{et al.}, these yield eighteen pure $q$-ary quantum error-correcting codes across $q \in \{2, 3, 4, 5, 7\}$, including seven new distance-$7$ codes:
$[[33, 11, 7]]_2$, $[[34, 12, 7]]_2$, $[[36, 26, 4]]_2$, $[[49, 33, 5]]_2$, $[[20, 10, 5]]_3$, $[[18, 6, 6]]_4$, $[[19, 7, 6]]_4$, $[[20, 10, 5]]_4$, $[[24, 8, 7]]_4$, $[[25, 9, 7]]_4$, $[[25, 15, 5]]_4$, $[[26, 16, 5]]_4$, $[[19, 7, 6]]_5$, $[[24, 12, 6]]_5$, $[[32, 16, 7]]_5$, $[[33, 17, 7]]_5$, $[[21, 7, 7]]_7$, and $[[22, 8, 7]]_7$.
Together with trivial stabilizer extension and pure subcode propagation, these constructions strictly improve \textbf{twenty-nine} minimum-distance lower bounds in Grassl's online tables (\texttt{CodeTables.de}), closing \textbf{five} previously open entries ($[[36, 26, 4]]_2$, $[[37, 26, 4]]_2$, $[[20, 10, 5]]_3$, $[[25, 15, 5]]_4$, and $[[26, 16, 5]]_4$) to exact optimality. Every code is specified completely in closed polynomial form and verified from first principles by standalone zero-symmetry verifiers in Python, Rust, and Magma.
\end{abstract}

\section{Introduction}

Let $\mathbb{F}_q$ be the finite field with $q$ elements ($q$ a prime power). An $[[n, k, d]]_q$ quantum stabilizer code encodes $k$ logical $q$-ary qudits into $n$ physical qudits and corrects any error affecting at most $\lfloor (d-1)/2 \rfloor$ qudits (or detects up to $d-1$ errors)~\cite{Calderbank1998,Ashikhmin2001,Ketkar2006}. Determining the largest achievable minimum distance $d$ for given $(q, n, k)$ is a central problem in quantum coding theory; the current state of knowledge is maintained by Grassl in the reference database \texttt{CodeTables.de}~\cite{Grassl2007}.

By the \emph{Hermitian construction}~\cite{Ashikhmin2001,Ketkar2006}, let $\mathbb{F}_{q^2}$ carry the Frobenius involution $\overline{u} = u^q$ and the Hermitian inner product $\langle \mathbf{u}, \mathbf{v} \rangle_H = \sum_{j=0}^{n-1} u_j v_j^q \in \mathbb{F}_{q^2}$. If $C \subseteq \mathbb{F}_{q^2}^n$ is an $\mathbb{F}_{q^2}$-linear $[n, m]_{q^2}$ code that is \emph{Hermitian self-orthogonal} ($C \subseteq C^{\perp_H}$) with $d(C^{\perp_H}) \ge d_{\mathrm{new}}$, then there exists a \emph{pure} $[[n, n - 2m, \ge d_{\mathrm{new}}]]_q$ stabilizer code~\cite[Corollary~19]{Ketkar2006}.

Extending our initial report (\texttt{v1.0.0}, ten codes / sixteen entries), we incorporate norm-$1$ $\lambda$-constacyclic multipliers ($\lambda^{q+1} = 1$) and general $\beta$-borders ($\beta \in \mathbb{F}_{q^2}^\times$), obtaining eight additional explicit pure quantum codes (seven of distance $d = 7$, including binary qubit codes $[[33, 11, 7]]_2$ and $[[34, 12, 7]]_2$). In total, these eighteen explicit codes improve twenty-nine entries in \texttt{CodeTables.de}.

\section{Constacyclic and Bordered Hermitian Self-Orthogonality}

Fix a block size $m \ge 3$ and a norm-$1$ scalar $\lambda \in \mathbb{F}_{q^2}^\times$ satisfying $\lambda \overline{\lambda} = \lambda^{q+1} = 1$, and let $R_{m,\lambda} = \mathbb{F}_{q^2}[x]/(x^m - \lambda)$. With each polynomial $a(x) = \sum_{t=0}^{m-1} a_t x^t \in R_{m,\lambda}$ we associate the $m \times m$ $\lambda$-constacyclic matrix $M_\lambda(a)$ whose $(r, c)$ entry ($0 \le r, c < m$) is $a_{r - c}$ for $r \ge c$ and $\lambda a_{m + r - c}$ for $r < c$ (so column $0$ is $(a_0, a_1, \dots, a_{m-1})^\top$). Since $\overline{\lambda} = \lambda^{-1}$, defining the involution
\[
  a^\dagger(x) = a_0^q + \overline{\lambda} \sum_{t=1}^{m-1} a_{m-t}^q x^t \in R_{m,\lambda}
\]
ensures $M_\lambda(a)^\dagger = \overline{M_\lambda(a)}^\top = M_\lambda(a^\dagger)$ and $M_\lambda(a) M_\lambda(b) = M_\lambda(a b \bmod (x^m - \lambda))$.

\begin{proposition}[Quasi-Twisted and Bordered Hermitian Construction]\label{prop:hermitian}
Let $\ell \ge 3$, let $a_1(x), \dots, a_{\ell-1}(x) \in R_{m,\lambda}$ with $\lambda^{q+1} = 1$, and let $\mathbf{1}_m = (1, \dots, 1)^\top \in \mathbb{F}_{q^2}^m$.
\begin{enumerate}
  \item \emph{(Unbordered, $n = \ell m$):} The $m \times \ell m$ matrix $G = [I_m \mid M_\lambda(a_1) \mid \cdots \mid M_\lambda(a_{\ell-1})]$ has rank $m$ over $\mathbb{F}_{q^2}$ and satisfies $G G^\dagger = \mathbf{0}$ if and only if
  \begin{equation}\label{eq:unbordered}
    \sum_{i=1}^{\ell-1} a_i(x)\, a_i^\dagger(x) \equiv -1 \pmod{x^m - \lambda}.
  \end{equation}
  \item \emph{(Bordered at $\lambda = 1$, $n = \ell m + 1$):} For $\lambda = 1$ and $\beta \in \mathbb{F}_{q^2}^\times$, the $m \times (\ell m + 1)$ matrix $G_{\mathrm{bord}} = [I_m \mid M_1(a_1) \mid \cdots \mid M_1(a_{\ell-1}) \mid \beta \mathbf{1}_m]$ has rank $m$ and satisfies $G_{\mathrm{bord}} G_{\mathrm{bord}}^\dagger = \mathbf{0}$ if and only if
  \begin{equation}\label{eq:bordered}
    \sum_{i=1}^{\ell-1} a_i(x)\, a_i^\dagger(x) \equiv -1 - \beta^{q+1} \sum_{t=0}^{m-1} x^t \pmod{x^m - 1}.
  \end{equation}
\end{enumerate}
\end{proposition}

\begin{proof}
Since $G$ begins with $I_m$, $\mathrm{rank}_{\mathbb{F}_{q^2}}(G) = m$. For~(1), $G G^\dagger = M_\lambda\bigl(1 + \sum_{i=1}^{\ell-1} a_i a_i^\dagger\bigr) = \mathbf{0}$ iff~\eqref{eq:unbordered} holds. For~(2), $(\beta \mathbf{1}_m)(\beta \mathbf{1}_m)^\dagger = \beta^{q+1} J_m = M_1\bigl(\beta^{q+1} \sum_{t=0}^{m-1} x^t\bigr)$, giving~\eqref{eq:bordered}.
\end{proof}

Because $N_\lambda(a) = a(x) a^\dagger(x) \bmod (x^m - \lambda)$ satisfies $N_\lambda(a)^\dagger = N_\lambda(a)$, its values lie in an $\mathbb{F}_q$-subspace of size only $q^m$ inside $R_{m,\lambda}$ (of size $q^{2m}$). Hash-table norm matching therefore generates exact Hermitian self-orthogonal tuples in $O(1)$ amortized time, which are screened on an NVIDIA GeForce RTX 3090 GPU via cyclic anchor reduction and forced-pivot column cancellation ($d_{\mathrm{new}} \le 7$).

\section{Independent Clean-Room Verification and Data Availability}

(1)~our standalone zero-symmetry projective Meet-in-the-Middle verifier (\texttt{verify\_all\_qecc.py}), which checks SHA-256 digests, $\mathbb{F}_{q^2}$ RREF rank, $G G^\dagger = \mathbf{0}$, symplectic orthogonality $X Z^\top - Z X^\top \equiv \mathbf{0} \pmod q$, and exact minimum distance $d(C^{\perp_H}) \ge d_{\mathrm{new}}$ across all eighteen codes in $\sim 10$~seconds; and
